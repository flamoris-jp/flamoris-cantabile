add_executable(cantabile-render renderer.cpp)
target_compile_features(cantabile-render PRIVATE cxx_std_20)
target_link_libraries(cantabile-render PRIVATE ${CMAKE_DL_LIBS})
if(MSVC)
  target_compile_options(cantabile-render PRIVATE /W4 /EHsc)
else()
  target_compile_options(cantabile-render PRIVATE -Wall -Wextra -Wpedantic)
endif()
install(TARGETS cantabile-render RUNTIME DESTINATION bin)
