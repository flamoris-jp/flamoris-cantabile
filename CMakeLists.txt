cmake_minimum_required(VERSION 3.20)
project(FlamorisCantabile VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)

find_package(nlohmann_json 3.11 CONFIG QUIET)
if(NOT TARGET nlohmann_json::nlohmann_json)
    find_path(JSON_INCLUDE_DIR nlohmann/json.hpp REQUIRED)
    add_library(cantabile_json INTERFACE)
    target_include_directories(cantabile_json INTERFACE "${JSON_INCLUDE_DIR}")
    add_library(nlohmann_json::nlohmann_json ALIAS cantabile_json)
endif()

add_library(cantabile_core SHARED src/core.cpp src/midi.cpp)
target_include_directories(cantabile_core PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(cantabile_core PRIVATE nlohmann_json::nlohmann_json)
target_compile_definitions(cantabile_core PRIVATE CANTABILE_BUILDING_CORE)
set_target_properties(cantabile_core PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
if(MSVC)
    target_compile_options(cantabile_core PRIVATE /W4 /permissive-)
else()
    target_compile_options(cantabile_core PRIVATE -Wall -Wextra -Wpedantic)
endif()

add_executable(cantabile-cli src/cli.cpp)
target_link_libraries(cantabile-cli PRIVATE cantabile_core nlohmann_json::nlohmann_json)
if(UNIX AND NOT APPLE)
    set_target_properties(cantabile-cli PROPERTIES INSTALL_RPATH "$ORIGIN/../lib")
endif()

if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/audio/CMakeLists.txt")
    add_subdirectory(audio)
endif()

if(BUILD_TESTING)
    add_executable(cantabile-midi-tests tests/midi_codec_test.cpp src/midi.cpp)
    target_include_directories(cantabile-midi-tests PRIVATE include)
    target_link_libraries(cantabile-midi-tests PRIVATE nlohmann_json::nlohmann_json)
    # The fixtures use assert: keep every check active in Release/Windows CI.
    if(MSVC)
        target_compile_options(cantabile-midi-tests PRIVATE /UNDEBUG)
    else()
        target_compile_options(cantabile-midi-tests PRIVATE -UNDEBUG)
    endif()
    add_test(NAME midi-codec COMMAND cantabile-midi-tests)
endif()

install(TARGETS cantabile_core cantabile-cli RUNTIME DESTINATION bin LIBRARY DESTINATION lib ARCHIVE DESTINATION lib)
install(FILES include/cantabile/core.h DESTINATION include/cantabile)
